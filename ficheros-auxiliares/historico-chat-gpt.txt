2. Proceso principal: sf-cierre-fase1-develop
Qué hace
Este proceso corresponde al proyecto UiPath Datos Cierre.

Su función es extraer de Salesforce un informe de oportunidades cerradas/ganadas, exportarlo a Excel, filtrar las oportunidades con WBS válido y generar items en la cola de Orchestrator CierreOP.

Para tu caso, el dato más importante que debería obtener es el WBS/PEP asociado a la oportunidad 20250256445.

Por qué es el proceso que necesitas
En el CORE dejamos pendiente, entre otros, este dato:

var2_SAP_PEP_WBS
Aunque lo habíamos etiquetado como dato SAP, en el flujo real descrito por los responsables parece que el WBS/PEP viaja desde Salesforce o desde el proceso previo de cierre. Este proceso es el que lee precisamente el campo WBS de Salesforce.

Configuración previa necesaria
Abre en UiPath Studio el proyecto:

sf-cierre-fase1-develop/project.json
El entrypoint principal es:

Main.xaml
Assets de Orchestrator esperados
El proceso espera encontrar estos assets:

Asset	Uso
PathRoot	Carpeta raíz local donde el robot trabaja.
PathOutput	Ruta base de salida/repositorio. Si vale Usuario, usa la carpeta de usuario de Windows/Getronics.
Informe-cierre	Nombre del informe Salesforce que debe abrir/exportar.
mail-from	Remitente para notificación.
mail-to	Destinatarios para notificación.
Además, el proceso trabaja con la cola:

CierreOP
Y usa por defecto el nombre de robot:

Getronics
Por cómo está montado el flujo, debería existir esta estructura local:

<PathRoot>\Getronics\
<PathRoot>\Getronics\excel\

Muy importante: el proceso intenta copiar una plantilla desde:

Si esa plantilla no existe, el proceso fallará al crear el Excel de salida. Debes localizarla en GitLab, en el entorno del robot o crear una plantilla compatible con la hoja Oportunidades.

Informe Salesforce que debe usar
El asset Informe-cierre debe apuntar a un informe de Salesforce que contenga, como mínimo, estas columnas:

Si cambia el orden del informe Salesforce, puede coger una columna incorrecta. Antes de confiar en fase 2, valida el Excel exportado y comprueba qué valor se está metiendo en Correo.

 └─ Getronics

void ReleaseCom(object obj)
{
    if (obj != null)

        try
        {
            System.Runtime.InteropServices.Marshal.ReleaseComObject(obj);
        }

string Norm(object value)
{
    if (value == null) return string.Empty;
    return System.Convert.ToString(value).Trim();
}

double? ToNumber(object value)
{
    if (value == null) return null;

    if (value is double) return (double)value;
    if (value is float) return System.Convert.ToDouble(value);
    if (value is int || value is long || value is decimal) return System.Convert.ToDouble(value);

    string txt = Norm(value);
    if (string.IsNullOrWhiteSpace(txt)) return null;

    double n;

    if (double.TryParse(txt, System.Globalization.NumberStyles.Any, System.Globalization.CultureInfo.CurrentCulture, out n))
        return n;

    if (double.TryParse(txt, System.Globalization.NumberStyles.Any, System.Globalization.CultureInfo.GetCultureInfo("es-ES"), out n))
        return n;

    if (double.TryParse(txt, System.Globalization.NumberStyles.Any, System.Globalization.CultureInfo.InvariantCulture, out n))
        return n;

    string normalized = txt.Replace(".", "").Replace(",", ".");

    if (double.TryParse(normalized, System.Globalization.NumberStyles.Any, System.Globalization.CultureInfo.InvariantCulture, out n))
        return n;

    return null;
}

int ToInt(object value, int defaultValue)
{
    double? n = ToNumber(value);
    if (!n.HasValue) return defaultValue;

    return System.Convert.ToInt32(System.Math.Round(n.Value, 0));
}

System.DateTime? ToDate(object value)
{
    if (value == null) return null;

    if (value is System.DateTime) return (System.DateTime)value;

    double? n = ToNumber(value);
    if (n.HasValue && n.Value > 1000)
        return System.DateTime.FromOADate(n.Value);

    string txt = Norm(value);
    if (string.IsNullOrWhiteSpace(txt)) return null;

    System.DateTime dt;

    if (System.DateTime.TryParse(txt, System.Globalization.CultureInfo.GetCultureInfo("es-ES"), System.Globalization.DateTimeStyles.None, out dt))
        return dt;

    if (System.DateTime.TryParse(txt, out dt))
        return dt;

    return null;
}

string ColumnLetter(int columnNumber)
{
    int dividend = columnNumber;
    string columnName = string.Empty;

    while (dividend > 0)
    {
        int modulo = (dividend - 1) % 26;
        columnName = System.Convert.ToChar(65 + modulo).ToString() + columnName;
        dividend = (dividend - modulo) / 26;
    }

    return columnName;
}

int ExtractYear(object value)
{
    string txt = Norm(value);
    var match = System.Text.RegularExpressions.Regex.Match(txt, @"20\d{2}");

    if (match.Success)
        return int.Parse(match.Value);

    double? n = ToNumber(value);

    if (n.HasValue && n.Value >= 2000 && n.Value <= 2100)
        return System.Convert.ToInt32(n.Value);

    return 0;
}

void AddToYearMap(System.Collections.Generic.Dictionary<int, double> map, int year, object value)
{
    double? n = ToNumber(value);

    if (n.HasValue && System.Math.Abs(n.Value) > 0.0000001)
    {
        if (!map.ContainsKey(year))
            map[year] = 0.0;

        map[year] = map[year] + n.Value;
    }
}

System.Collections.Generic.List<System.DateTime> GetMonths(System.DateTime startDate, int duration)

    var months = new System.Collections.Generic.List<System.DateTime>();

    for (int i = 0; i < duration; i++)
    {

        months.Add(new System.DateTime(d.Year, d.Month, 1));
    }

    return months;
}

System.Collections.Generic.List<double?> GetDistribution(
    System.Collections.Generic.Dictionary<int, double> byYear,
    System.Collections.Generic.List<System.DateTime> months)
{
    var counts = new System.Collections.Generic.Dictionary<int, int>();

    foreach (var m in months)
    {
        if (!counts.ContainsKey(m.Year))
            counts[m.Year] = 0;

        counts[m.Year] = counts[m.Year] + 1;
    }

    var values = new System.Collections.Generic.List<double?>();

    foreach (var m in months)
    {
        if (byYear.ContainsKey(m.Year) && counts.ContainsKey(m.Year) && counts[m.Year] > 0)
            values.Add(byYear[m.Year] / counts[m.Year]);
        else
            values.Add(null);
    }

    return values;
}

System.Collections.Generic.Dictionary<string, double> ReadTariffs(dynamic sintesis)
{
    var numerators = new System.Collections.Generic.Dictionary<string, double>(System.StringComparer.OrdinalIgnoreCase);
    var denominators = new System.Collections.Generic.Dictionary<string, double>(System.StringComparer.OrdinalIgnoreCase);
    var directRates = new System.Collections.Generic.Dictionary<string, double>(System.StringComparer.OrdinalIgnoreCase);

    if (sintesis == null)
        return directRates;

    for (int r = 18; r <= 80; r++)
    {
        string sigla = Norm(GetCell(sintesis, "D" + r));

        if (string.IsNullOrWhiteSpace(sigla) || sigla == "0")
            continue;

        double? hours = ToNumber(GetCell(sintesis, "N" + r));
        double? tariff = ToNumber(GetCell(sintesis, "O" + r));
        double? sale = ToNumber(GetCell(sintesis, "P" + r));
        double? effectiveRate = null;

        if (tariff.HasValue && System.Math.Abs(tariff.Value) > 0.0000001)
            effectiveRate = tariff.Value;
        else if (sale.HasValue && hours.HasValue && System.Math.Abs(hours.Value) > 0.0000001)
            effectiveRate = sale.Value / hours.Value;

        if (!effectiveRate.HasValue || System.Math.Abs(effectiveRate.Value) < 0.0000001)
            continue;

        if (hours.HasValue && System.Math.Abs(hours.Value) > 0.0000001)
        {
            if (!numerators.ContainsKey(sigla))
            {
                numerators[sigla] = 0.0;
                denominators[sigla] = 0.0;
            }

            numerators[sigla] = numerators[sigla] + effectiveRate.Value * hours.Value;
            denominators[sigla] = denominators[sigla] + hours.Value;
        }
        else if (!directRates.ContainsKey(sigla))
        {
            directRates[sigla] = effectiveRate.Value;
        }
    }

    foreach (var kv in numerators)
    {
        if (denominators.ContainsKey(kv.Key) && System.Math.Abs(denominators[kv.Key]) > 0.0000001)
            directRates[kv.Key] = kv.Value / denominators[kv.Key];
    }

    return directRates;
}

        {
            { "Project", project },
            { "Financial", financial },
            { "Profiles", profileList },
            { "Expenses", expenses },
            { "Purchases", purchases }
        };
    }
    finally
    {
        ReleaseCom(hd);
        ReleaseCom(pre);
        ReleaseCom(par);
        ReleaseCom(fsap);
        ReleaseCom(sintesis);
    }
}

System.Collections.Generic.Dictionary<string, object> FindCierreSfRow(dynamic workbook, string opportunityNumber)
{
    foreach (object wsObj in workbook.Worksheets)
    {
        dynamic ws = wsObj;
        object usedObj = null;

        try
        {
            usedObj = ws.UsedRange;
            dynamic used = usedObj;

            int rowCount = System.Convert.ToInt32(used.Rows.Count);
            int colCount = System.Convert.ToInt32(used.Columns.Count);
            int maxHeaderRow = System.Math.Min(10, rowCount);

            for (int headerRow = 1; headerRow <= maxHeaderRow; headerRow++)
            {
                var headers = new System.Collections.Generic.Dictionary<string, int>(System.StringComparer.OrdinalIgnoreCase);

                for (int c = 1; c <= colCount; c++)
                {
                    string h = Norm(ws.Cells.Item(headerRow, c).Value2);

                    if (!string.IsNullOrWhiteSpace(h) && !headers.ContainsKey(h))
                        headers[h] = c;
                }

                int oppCol = 0;

                string[] oppNames = new string[]
                {
                    "Opportunity Number",
                    "NumeroSF",
                    "NúmeroSF",
                    "OpportunityNumber",
                    "SLFC",
                    "Opportunity Code"
                };

                foreach (string name in oppNames)
                {
                    if (headers.ContainsKey(name))
                    {
                        oppCol = headers[name];
                        break;
                    }
                }

                if (oppCol == 0)
                    continue;

                for (int r = headerRow + 1; r <= rowCount; r++)
                {
                    string value = Norm(ws.Cells.Item(r, oppCol).Value2);

                    if (string.Equals(value, opportunityNumber, System.StringComparison.OrdinalIgnoreCase))
                    {
                        var row = new System.Collections.Generic.Dictionary<string, object>(System.StringComparer.OrdinalIgnoreCase);

                        foreach (var header in headers)
                            row[header.Key] = ws.Cells.Item(r, header.Value).Value2;

                        row["__SheetName"] = System.Convert.ToString(ws.Name);
                        row["__Row"] = r;

                        return row;
                    }
                }
            }
        }
        finally
        {
            ReleaseCom(usedObj);
            ReleaseCom(wsObj);
        }
    }

    return null;
}

string GetAny(System.Collections.Generic.Dictionary<string, object> row, params string[] names)
{
    if (row == null) return string.Empty;

    foreach (string n in names)
    {
        if (row.ContainsKey(n))
        {
            string val = Norm(row[n]);

            if (!string.IsNullOrWhiteSpace(val))
                return val;
        }
    }

    return string.Empty;
}

void WriteResourceRows(

    System.Collections.Generic.List<System.Collections.Generic.Dictionary<string, object>> profiles,
    System.DateTime startDate,
    int duration,
    bool isAT)
{
    if (isAT)
    {
        if (profiles.Count > 15)
            throw new System.Exception(
                "La plantilla CORE AT incluida soporta 15 perfiles AT en Resources/Cost Planning. Perfiles encontrados: "
                + profiles.Count.ToString()
                + ".");

        ClearRange(sheetRes, "B7:E36");
        ClearRange(sheetRes, "G7:AF36");

        int resourceStartCol = 7;
        int rowIndex = 7;

            SetCell(sheetRes, "C" + rowIndex, sig);

                SetCell(sheetRes, col + (rowIndex + 1), tariff.HasValue ? (object)tariff.Value : null);
            }

            rowIndex += 2;
            varIndex++;
        }
    }
    else
    {
        ClearRange(sheetRes, "B7:AD43");

        int resourceStartCol = 6;
        int rowIndex = 7;

        foreach (var p in profiles)
        {
            string sig = Norm(p["Sigla"]);
            string interco = GetInterco(project, p);
            string employeeVar = "var" + (rowIndex + 3).ToString() + "_SAP_CodEmpleado_" + sig;

            SetCell(sheetRes, "B" + rowIndex, employeeVar);
            SetCell(sheetRes, "C" + rowIndex, sig);
            SetCell(sheetRes, "D" + rowIndex, interco);
            SetCell(sheetRes, "E" + rowIndex, p["Profile"]);

            for (int i = 0; i < duration; i++)

                string col = ColumnLetter(resourceStartCol + i);
                SetCell(sheetRes, col + rowIndex, baseCost.HasValue ? (object)baseCost.Value : null);
            }

            rowIndex++;
        }
    }
}

void WriteCoreWorkbook(
    dynamic excel,
    string templatePath,
    string outputPath,
    System.Collections.Generic.Dictionary<string, object> ppoData,
    System.Collections.Generic.Dictionary<string, object> cierreRow)
{
    System.IO.File.Copy(templatePath, outputPath, true);

    dynamic wb = null;
    dynamic sheetPI = null;
    dynamic sheetRes = null;
    dynamic sheetCP = null;
    dynamic sheetCO = null;

    try
    {
        wb = excel.Workbooks.Open(outputPath, 0, false);

        var project = (System.Collections.Generic.Dictionary<string, object>)ppoData["Project"];
        var financial = (System.Collections.Generic.Dictionary<string, object>)ppoData["Financial"];
        var profiles = (System.Collections.Generic.List<System.Collections.Generic.Dictionary<string, object>>)ppoData["Profiles"];
        var expenses = (System.Collections.Generic.List<System.Collections.Generic.Dictionary<string, object>>)ppoData["Expenses"];
        var purchases = (System.Collections.Generic.List<System.Collections.Generic.Dictionary<string, object>>)ppoData["Purchases"];

        var startDate = (System.DateTime)project["StartDate"];
        int duration = System.Convert.ToInt32(project["Duration"]);
        var months = GetMonths(startDate, duration);
        bool isAT = IsAtProject(project);

        string wbs = GetAny(cierreRow, "WBS", "Wbs", "PEP", "WBS/PEP", "Código Proyecto", "Codigo Proyecto");
        string accountName = GetAny(cierreRow, "AccountName", "Account Name", "Cuenta", "Cliente");
        string sfPractice = GetAny(cierreRow, "Practica", "Práctica", "Practice");
        string pmName = GetAny(cierreRow, "Project Manager", "Jefe Proyecto", "Jefe de Proyecto", "PM");
        string salesOrder = GetAny(cierreRow, "Sales Order", "SalesOrder", "Pedido", "Pedido SAP", "Order");
        string customerCode = GetAny(cierreRow, "Customer Code", "Codigo Cliente", "Código Cliente", "Sold-to number", "SoldTo", "Sold-to");
        string salesManagerCode = GetAny(cierreRow, "Sales Manager ID", "SalesManagerID", "Cod Sales Manager", "Código Sales Manager");

        sheetPI = GetWorksheet(wb, "Project Infor", true);
        sheetRes = GetWorksheet(wb, "Resources", true);
        sheetCP = GetWorksheet(wb, "Cost Planning", true);
        sheetCO = GetWorksheet(wb, "Cost Overview", false);

        SetCell(sheetPI, "C5", System.DateTime.Today);
        SetCell(sheetPI, "C8", project["StartDate"]);
        SetCell(sheetPI, "C9", project["EndDate"]);
        SetCell(sheetPI, "C10", !string.IsNullOrWhiteSpace(wbs) ? (object)wbs : "var2_SAP_PEP_WBS");
        SetCell(sheetPI, "C11", project["Description"]);
        SetCell(sheetPI, "C12", !string.IsNullOrWhiteSpace(sfPractice) ? (object)sfPractice : project["Practice"]);
        SetCell(sheetPI, "C13", "");
        SetCell(sheetPI, "C16", !string.IsNullOrWhiteSpace(customerCode) ? (object)customerCode : "var3_SAP_CodigoCliente");
        SetCell(sheetPI, "D16", !string.IsNullOrWhiteSpace(accountName) ? (object)accountName : project["Customer"]);
        SetCell(sheetPI, "C18", "SLFC " + Norm(project["OpportunityCode"]));
        SetCell(sheetPI, "C19", !string.IsNullOrWhiteSpace(salesOrder) ? (object)salesOrder : "var4_SAP_PedidoSalesOrder");
        SetCell(sheetPI, "C22", "var5_SAP_CodEmpleadoPM");
        SetCell(sheetPI, "D22", !string.IsNullOrWhiteSpace(pmName) ? (object)pmName : "var6_SAP_ProjectManager");
        SetCell(sheetPI, "C23", "var7_SAP_CodEmpleadoPracticeLeader");
        SetCell(sheetPI, "D23", "var8_SAP_PracticeLeader");
        SetCell(sheetPI, "C24", !string.IsNullOrWhiteSpace(salesManagerCode) ? (object)salesManagerCode : "var9_SF_CodEmpleadoSalesManager");
        SetCell(sheetPI, "D24", project["AccountManager"]);
        SetCell(sheetPI, "C28", financial["OrderAmount"]);
        SetCell(sheetPI, "C29", financial["HoursEstimated"]);
        SetCell(sheetPI, "C30", financial["ResourceCost"]);
        SetCell(sheetPI, "C31", financial["Risks"]);

        if (financial["Warranty"] != null && System.Math.Abs(System.Convert.ToDouble(financial["Warranty"])) > 0.0000001)
            SetCell(sheetPI, "C32", financial["Warranty"]);
        else
            SetCell(sheetPI, "C32", "");

        SetCell(sheetPI, "C33", financial["Expenses"]);
        SetCell(sheetPI, "C34", financial["Purchases"]);

        WriteResourceRows(sheetRes, project, profiles, startDate, duration, isAT);
        WritePlanningRows(sheetCP, project, financial, profiles, expenses, purchases, months, isAT);

        if (sheetCO != null)
        {
            SetCell(sheetCO, "G17", "");
            SetCell(sheetCO, "I17", "");
        }

        try
        {
            excel.CalculateFullRebuild();
        }
        catch
        {
        }

        try
        {
            wb.ForceFullCalculation = true;
            wb.FullCalculationOnLoad = true;
        }
        catch
        {
        }

        wb.SaveAs(outputPath, 51);
    }
    finally
    {
        ReleaseCom(sheetPI);
        ReleaseCom(sheetRes);
        ReleaseCom(sheetCP);
        ReleaseCom(sheetCO);

        if (wb != null)
        {
            try
            {
                wb.Close(true);
            }
            catch
            {
            }

            ReleaseCom(wb);
        }
    }
}
